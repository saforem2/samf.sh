export const allPosts = (await getCollection('docs'))
    .filter((doc) => {
        const id = doc.id
        if (!id) return false
        return (
            id.startsWith('posts/') &&
            !id.startsWith('posts/drafts/') &&
            !id.endsWith('posts/index.mdx') &&
            !id.match(/^posts\/\d{4}\/index\.mdx$/) &&
            !id.match(/^posts\/\d{4}\/\d{2}\/index\.mdx$/) &&
            !id.match(/^posts\/(auroragpt|ai-for-physics|jupyter)\/index\.mdx$/) &&
            doc.data.date &&
            !doc.data.draft
        )
    })
    .sort((a, b) => new Date(b.data.date).getTime() - new Date(a.data.date).getTime())

export const allTalks = (await getCollection('docs'))
    .filter((doc) => {
        const id = doc.id
        if (!id) return false
        return (
            id.startsWith('talks/') &&
            id !== 'talks/index.mdx' &&
            doc.data.date &&
            !doc.data.draft
        )
    })
    .sort((a, b) => new Date(b.data.date).getTime() - new Date(a.data.date).getTime())

  function updateNowPlayingCard(track) {
    const trackTitle = document.getElementById('tracktitle')

    if (!trackTitle || !trackArtist || !trackArt) return

  async function fetchLatestTrack() {
    const params = new URLSearchParams({
      user: lastFmConfig.user,
      api_key: lastFmConfig.apiKey,
      format: 'json',
      limit: '1'
    })

    const response = await fetch(lastFmConfig.apiBase + '&' + params.toString())
    if (!response.ok) {
      throw new Error('Last.fm API error: ' + response.status)
    }

    const data = await response.json()
    const recentTrack = data?.recenttracks?.track?.[0]

  /* The posts / talks / work lists run 37, 23 and 27 rows. Shown in full they
     bury everything below them, so each is capped at 20 lines and scrolls.
     The height goes on .table-scroll-wrapper (Doc.astro wraps every table in
     one at runtime) rather than the table itself: max-height on a <table> is
     only a hint and gets ignored once the rows exceed it. */
  .table-scroll-wrapper:has(> #posts-table),
  .table-scroll-wrapper:has(> #talks-table),
  .table-scroll-wrapper:has(> #repo-table),
  #work-table .table-scroll-wrapper {
    max-height: 20lh;
    overflow-y: auto;
    /* The base .table-scroll-wrapper rule sets width: 100%, which left the
       posts box 835px wide around a 626px table: 208px of dead space with the
       scrollbar stranded at the far edge. fit-content shrinks the box to the
       table while max-width still clamps the wide case (talks is 887px and
       genuinely scrolls). */
    width: fit-content;
    /* Reserve the scrollbar track so the table does not shift sideways when
       the list is short enough not to need one. */
    scrollbar-gutter: stable;
    /* Leave a gutter that belongs to the PAGE, not the table, so there is
       somewhere to put the pointer that scrolls the page normally. Measured
       without it: a wheel over a capped table fed all 720px into the inner
       list and advanced the page 0, so scrolling past the posts list took ~8
       ticks before the page moved at all. overscroll-behavior only governs
       the inner boundary, so it cannot solve the pass-through on its own.
       End side only: a leading margin indented the tables out of line with
       the body text. */
    margin-inline-end: 8ch;
    max-width: calc(100% - 8ch);
  }
  /* No gutter on narrow screens: 8ch costs a third of a 358px column there,
     and touch scrolling does not have the pointer-capture problem the gutter
     exists to solve. */
  @media (max-width: 80ch) {
    .table-scroll-wrapper:has(> #posts-table),
    .table-scroll-wrapper:has(> #talks-table),
    .table-scroll-wrapper:has(> #repo-table),

      margin-inline-end: 0;
      max-width: 100%;
    }
  }
  /* Keep the column headers visible while scrolling. WebTUI sets
     position: relative on every td, and a positioned element paints above a
     non-positioned one, so the pinned header needs an explicit z-index on the
